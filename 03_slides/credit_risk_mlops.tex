\begin{frame}{Modelos y TPE}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Modelos evaluados}
\begin{itemize}
  \item Regresión logística, árbol, Gaussian NB y SVM RBF.
  \item Random Forest, Extra Trees, XGBoost y LightGBM.
  \item MLP de PyTorch.
\end{itemize}
\textbf{Referencias}
\begin{itemize}
  \item Baseline heurístico.
  \item Dummy: estrategia \texttt{constant}, clase fija 1 (pago a tiempo).
\end{itemize}
\column{0.48\textwidth}
\textbf{Búsqueda TPE}
\scriptsize
\begin{itemize}
  \item 30 intentos por modelo.
  \item Seed TPE 20260910: generación pseudoaleatoria de intentos.
  \item Logística: $C$ y pesos; Gaussian NB: \texttt{var\_smoothing}; SVM: $C$, \texttt{gamma} y pesos.
  \item Árboles: profundidad, hojas, estimadores, variables por split y pesos.
  \item Gradient boosting (XGBoost/LightGBM): estimadores, profundidad/hojas, tasa de aprendizaje, muestreo, columnas, regularización y balance.
  \item MLP: capas, tasa de aprendizaje, dropout, lote, épocas y regularización.
\end{itemize}
\end{columns}
\end{frame}
