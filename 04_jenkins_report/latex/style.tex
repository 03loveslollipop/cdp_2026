% Full-width evidence, with the caption in the right annotation column below.
% Reserve its true height so the next image/table cannot cover the caption.
\newcommand{\wideevidence}[3]{%
  \par\medskip\begin{fullwidth}\noindent
  \begin{minipage}{\linewidth}\centering
    \includegraphics[width=\linewidth,height=.49\textheight,keepaspectratio]{#1}\par
    \smallskip\noindent\hfill
    \begin{minipage}{\marginparwidth}
      \captionof{figure}{\strut}\label{#3}
    \end{minipage}
  \end{minipage}\end{fullwidth}\par\medskip}
